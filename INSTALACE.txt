CREATE TABLE treneri (
    id INT AUTO_INCREMENT PRIMARY KEY,
    jmeno VARCHAR(100) NOT NULL,
    email VARCHAR(255) NOT NULL,
    heslo VARCHAR(100) NOT NULL,
    role ENUM('trener','hlavni') DEFAULT 'trener'
);

CREATE TABLE skupiny (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nazev VARCHAR(100) NOT NULL,
    poradi INT DEFAULT 0
);

CREATE TABLE podskupiny (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nazev VARCHAR(100) NOT NULL,
    skupina_id INT NOT NULL,
    poradi INT DEFAULT 0,
    FOREIGN KEY (skupina_id) REFERENCES skupiny(id)
);

CREATE TABLE treninky (
    id INT AUTO_INCREMENT PRIMARY KEY,
    trener_id INT NOT NULL,
    datum DATE NOT NULL,
    napln TEXT NOT NULL,
    poznamka TEXT,
    delka DECIMAL(5,2) DEFAULT 0,
    obrazky TEXT,
    mereni TEXT,
    FOREIGN KEY (trener_id) REFERENCES treneri(id)
);

CREATE TABLE trenink_skupina (
    trenink_id INT NOT NULL,
    skupina_id INT NOT NULL,
    PRIMARY KEY (trenink_id, skupina_id),
    FOREIGN KEY (trenink_id) REFERENCES treninky(id),
    FOREIGN KEY (skupina_id) REFERENCES skupiny(id)
);

CREATE TABLE trenink_podskupina (
    trenink_id INT NOT NULL,
    podskupina_id INT NOT NULL,
    PRIMARY KEY (trenink_id, podskupina_id),
    FOREIGN KEY (trenink_id) REFERENCES treninky(id),
    FOREIGN KEY (podskupina_id) REFERENCES podskupiny(id)
);

CREATE TABLE sportovci (
    id INT AUTO_INCREMENT PRIMARY KEY,
    jmeno VARCHAR(100) NOT NULL,
    hash VARCHAR(64) NOT NULL
);

CREATE TABLE trenink_sportovec (
    trenink_id INT NOT NULL,
    sportovec_id INT NOT NULL,
    PRIMARY KEY (trenink_id, sportovec_id),
    FOREIGN KEY (trenink_id) REFERENCES treninky(id),
    FOREIGN KEY (sportovec_id) REFERENCES sportovci(id)
);

CREATE TABLE dalsi_cinnosti (
    id INT AUTO_INCREMENT PRIMARY KEY,
    trener_id INT NOT NULL,
    nazev VARCHAR(255) NOT NULL,
    delka DECIMAL(5,2) DEFAULT 0,
    poznamka TEXT,
    datum DATE NOT NULL,
    FOREIGN KEY (trener_id) REFERENCES treneri(id)
);

CREATE TABLE story_nastaveni (
    typ ENUM('skupina','podskupina') NOT NULL,
    entita_id INT NOT NULL,
    barva VARCHAR(10),
    hlavicka VARCHAR(255),
    paticka VARCHAR(255),
    barva_textu VARCHAR(10),
    logo VARCHAR(255),
    PRIMARY KEY (typ, entita_id)
);

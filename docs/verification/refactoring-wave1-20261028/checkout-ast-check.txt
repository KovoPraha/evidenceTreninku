function normalized($node) {
    if ($node instanceof PhpParser\Node\Expr\BinaryOp\Concat || $node instanceof PhpParser\Node\Scalar\String_ || $node instanceof PhpParser\Node\Scalar\InterpolatedString) {
        $parts = stringParts($node); $merged=[];
        foreach ($parts as $p) {
            if (is_string($p) && is_string(end($merged))) $merged[count($merged)-1].=$p;
            else $merged[]=$p;
        }
        return ['stringParts'=>$merged];
    }
    if ($node instanceof PhpParser\Node) {
        $r=['type'=>$node->getType()];
        foreach ($node->getSubNodeNames() as $key) $r[$key]=normalized($node->$key);
        return $r;
    }
    return is_array($node)?array_map('normalized',$node):$node;
}
function stringParts($node) {
    if ($node instanceof PhpParser\Node\Expr\BinaryOp\Concat) return [...stringParts($node->left),...stringParts($node->right)];
    if ($node instanceof PhpParser\Node\Scalar\String_) return [$node->value];
    if ($node instanceof PhpParser\Node\Scalar\InterpolatedString) return array_map(fn($p)=>$p instanceof PhpParser\Node\InterpolatedStringPart?$p->value:normalized($p),$node->parts);
    return [normalized($node)];
}

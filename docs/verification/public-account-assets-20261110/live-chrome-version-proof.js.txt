async (page) => {
  const cdp=await page.context().newCDPSession(page);await cdp.send('Network.enable');
  const rows=[],errors=[],failed=[],optional=[];let css=new Map();
  page.on('pageerror',e=>errors.push(String(e)));
  page.on('console',m=>{if(m.type()==='error'){const u=m.location().url;if(u.endsWith('/favicon.ico')&&m.text().includes('404'))optional.push('optional favicon 404');else errors.push({text:m.text(),url:u});}});
  page.on('requestfailed',r=>failed.push({url:r.url(),error:r.failure()?.errorText}));
  cdp.on('Network.requestWillBeSent',e=>{if(e.request.url.includes('/bootstrap.min.css'))css.set(e.requestId,{url:e.request.url,id:e.requestId});});
  cdp.on('Network.responseReceived',e=>{if(css.has(e.requestId))Object.assign(css.get(e.requestId),{status:e.response.status,encoding:e.response.headers['content-encoding']||e.response.headers['Content-Encoding'],cache_control:e.response.headers['cache-control']||e.response.headers['Cache-Control']});});
  const hash=async value=>await page.evaluate(async s=>Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',new TextEncoder().encode(s)))).map(n=>n.toString(16).padStart(2,'0')).join(''),value);
  for(const path of ['/booking/prihlaseni.php','/booking/registrace.php','/booking/sportovec_prihlaseni.php','/booking/zapomenute_heslo.php','/booking/zapomenute_heslo.php?pro=trener','/booking/znovu_overit_email.php']) {
    await cdp.send('Network.setCacheDisabled',{cacheDisabled:true});await cdp.send('Network.clearBrowserCache');css=new Map();
    const response=await page.goto('https://kis.kovopraha.cz'+path,{waitUntil:'networkidle'});
    const raw=await response.text();
    const canonical=/<link href="\/(?!\/)[^"]*assets\/vendor\/bootstrap\/bootstrap\.min\.css\?v=[0-9]+" rel="stylesheet">/g;
    if((raw.match(canonical)||[]).length!==1)throw new Error('Canonical CSS missing '+path);
    const cdn=/^[ \t]*<link href="https:\/\/cdn\.jsdelivr\.net\/npm\/bootstrap@5\.3\.3\/dist\/css\/bootstrap\.min\.css" rel="stylesheet"\s+integrity="sha384-QWTKZyjpPEjISv5WaRU9OFeRpok6YctnYmDr5pNlyT2bRjXh0JMhjY6hW\+ALEwIH"\s+crossorigin="anonymous">[ \t]*(?:\r?\n)?/gm;
    let normalized=raw.replace(/\r\n/g,'\n').replace(cdn,'');
    const nonce=response.headers()['content-security-policy']?.match(/nonce-([^']+)/)?.[1];if(nonce)normalized=normalized.split(nonce).join('NONCE');
    normalized=normalized.replace(/(name="csrf_token"\s+value=")[^"]*(")/g,'$1CSRF$2');
    for(const item of css.values()){const payload=await cdp.send('Network.getResponseBody',{requestId:item.id});item.sha256=await hash(payload.body);delete item.id;}
    const replayedAssets=[];
    const replay=normalized.replace(/(\/assets\/(?:vendor\/bootstrap\/bootstrap\.min\.css|vendor\/bootstrap\/bootstrap\.bundle\.min\.js|vendor\/bootstrap-icons\/font\/bootstrap-icons\.css|app-ui\.css|app-ui\.js)\?v=)([0-9]+)/g,(all,prefix,version)=>{replayedAssets.push({path:prefix.split('?')[0],observed_after_version:version,observed_before_version:'1794302581'});return prefix+'1794302581';});
    if(replayedAssets.length!==5 || new Set(replayedAssets.map(a=>a.path)).size!==5)throw new Error('Exact five asset version replay required');
    const replayed_normalized_html_sha256=await hash(replay);
    rows.push({replayedAssets,replayed_normalized_html_sha256,path,status:response.status(),normalized_html_sha256:await hash(normalized),raw_html_bytes:Buffer.byteLength(raw),css:[...css.values()],browser:await page.evaluate(()=>({title:document.title,bootstrap:typeof window.bootstrap?.Modal==='function',resources:performance.getEntriesByType('resource').filter(e=>e.name.includes('/bootstrap.min.css')).map(e=>({url:e.name,encoded:e.encodedBodySize,transfer:e.transferSize,decoded:e.decodedBodySize})),fcp:performance.getEntriesByType('paint').find(e=>e.name==='first-contentful-paint')?.startTime}))});
  }
  await cdp.detach();return {complete:true,read_only:true,browser:page.context().browser().version(),rows,errors,failed,optional};
}

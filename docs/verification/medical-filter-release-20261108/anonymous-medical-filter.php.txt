<?php
declare(strict_types=1);
if (PHP_SAPI !== 'cli') { http_response_code(404); exit; }
$phase = $argv[1] ?? '';
if (!in_array($phase, ['before', 'after'], true)) { throw new RuntimeException('CLI before/after required.'); }
$url = 'https://kis.kovopraha.cz/medical_clearance_admin.php?medical_q=__read_performance_probe__&medical_from=2026-11-08&medical_to=2026-11-08&medical_state=valid';
$locationPath = null;
$handle = curl_init($url);
curl_setopt_array($handle, [CURLOPT_RETURNTRANSFER => true, CURLOPT_FOLLOWLOCATION => false,
    CURLOPT_CONNECTTIMEOUT => 15, CURLOPT_TIMEOUT => 45,
    CURLOPT_USERAGENT => 'KIS-authorized-read-verification',
    CURLOPT_HEADERFUNCTION => static function ($handle, string $line) use (&$locationPath): int {
        if (stripos($line, 'Location:') === 0) {
            $locationPath = parse_url(trim(substr($line, 9)), PHP_URL_PATH);
        }
        return strlen($line);
    }]);
$body = curl_exec($handle);
$status = curl_getinfo($handle, CURLINFO_RESPONSE_CODE);
$error = curl_error($handle);
curl_close($handle);
if ($body === false) { throw new RuntimeException($error); }
unset($body);
$pass = in_array($status, [401, 403], true)
    || (in_array($status, [301, 302, 303, 307, 308], true)
        && is_string($locationPath) && basename($locationPath) === 'login.php');
$result = ['at_utc' => gmdate('c'), 'url' => $url, 'method' => 'Anonymous GET without cookies, redirects or POST; response body and headers not stored.',
    'http' => $status, 'redirect_path' => $locationPath, 'pass' => $pass,
    'limitation' => 'Confirms anonymous rejection only. Authenticated production medical UI UAT was not performed.'];
file_put_contents(__DIR__ . '/anonymous-medical-filter-' . $phase . '.json', json_encode($result, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_THROW_ON_ERROR) . PHP_EOL);
echo json_encode($result, JSON_THROW_ON_ERROR) . PHP_EOL;
if (!$pass) { exit(1); }

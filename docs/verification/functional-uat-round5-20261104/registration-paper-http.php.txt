<?php

function r5Location(array $response,string $key):int{parse_str((string)parse_url($response['location'],PHP_URL_QUERY),$params);return (int)($params[$key]??0);}
function r5Term(int $id):array{return clubConsentTerm($GLOBALS['pdo'],$id);}
function r5Hash(array $row):string{return hash('sha256',json_encode($row,JSON_THROW_ON_ERROR));}
function r5PaperCount():int{return (int)$GLOBALS['pdo']->query('SELECT COUNT(*) FROM club_paper_records')->fetchColumn();}
function r5Dependencies(int $person):string{
 $p=$GLOBALS['pdo'];$q=$p->prepare('SELECT * FROM sportovci WHERE id=?');$q->execute([$person]);$data=['person'=>$q->fetch()];
 foreach(['account_person_roles','verejni_uzivatele','club_registration_consent_acceptances'] as $table)$data[$table]=$p->query('SELECT * FROM '.$table.' ORDER BY id')->fetchAll();
 return r5Hash($data);
}
$route='registration_texts_admin.php?custom=0';
dwCheck(dwRequest($registrar,$route)['status']===200,'Registrar opens custom consent editor');
dwCheck(dwRequest($coach,$route)['status']===403&&dwRequest($finance,$route)['status']===403,'Coach and finance cannot administer registration texts');
$title=$run.' souhlas <b>pouze text</b>';$text=$run.' syntetické znění. <b>Formátování se nespouští.</b>';
$create=['custom_action'=>'save','custom_id'=>0,'title'=>$title,'text'=>$text,'reason'=>$run.' vytvoření','custom_required'=>'1','custom_visible'=>'1'];
$before=(int)$pdo->query('SELECT COUNT(*) FROM club_registration_consents')->fetchColumn();
$bad=dwRequest($registrar,$route,['csrf_token'=>'forged']+$create);

$saved=dwRequest($registrar,$route,$create);$id=r5Location($saved,'custom');$draft=r5Location($saved,'draft_id');
dwCheck($saved['status']===303&&$id>0&&$draft>0,'Saving new custom consent creates a separate draft');
$current=r5Term($id);$initial=(int)$current['id'];
dwCheck((int)$current['custom_visible']===0,'New consent remains hidden until explicit publication');
dwCheck(!in_array($id,array_map('intval',array_column(clubConsentCurrent($pdo),'custom_consent_id')),true),'Unpublished consent is absent from public consent bundle');
$page=dwRequest($registrar,'registration_texts_admin.php?custom='.$id.'&draft_id='.$draft);
dwCheck(str_contains($page['body'],htmlspecialchars($title,ENT_QUOTES|ENT_SUBSTITUTE,'UTF-8'))&&!str_contains($page['body'],$title),'Saved title and markup are rendered as escaped plain text');
dwCheck(dwValue(dwDom($page),'title')===$title,'Editor preserves the exact submitted title');
$competing=dwRequest($registrar,$route,['custom_id'=>$id,'base_term_id'=>$initial,'text'=>$run.' druhý koncept']+$create);
$otherDraft=r5Location($competing,'draft_id');
dwCheck($competing['status']===303&&$otherDraft>$draft&&r5Term($id)['id']===$current['id'],'Competing draft does not change current public version');
$publish=['custom_action'=>'publish','custom_id'=>$id,'draft_id'=>$draft];
$noConfirm=dwRequest($registrar,$route,$publish);
dwCheck((int)r5Term($id)['id']===$initial&&str_contains($noConfirm['body'],'Výslovně potvrďte'),'Publication requires a separate explicit confirmation');
dwCheck(dwRequest($registrar,$route,['confirmed'=>'1']+$publish)['status']===303,'Explicit publication activates saved consent');
$published=r5Term($id);$publishedHash=r5Hash($published);
dwCheck((int)$published['custom_visible']===1&&(int)$published['custom_required']===1&&$published['consent_text_plain']===$text,'Published bundle preserves exact text and required flag');
dwCheck(in_array($id,array_map('intval',array_column(clubConsentCurrent($pdo),'custom_consent_id')),true),'Published consent appears in public consent bundle');
$stale=dwRequest($registrar,$route,['confirmed'=>'1','draft_id'=>$otherDraft]+$publish);
dwCheck(str_contains($stale['body'],'starší verze')&&r5Hash(r5Term($id))===$publishedHash,'Stale competing publication cannot overwrite current term');
$termCount=(int)$pdo->query('SELECT COUNT(*) FROM club_event_term_versions')->fetchColumn();
dwCheck(dwRequest($registrar,$route,['confirmed'=>'1']+$publish)['status']===303&&(int)$pdo->query('SELECT COUNT(*) FROM club_event_term_versions')->fetchColumn()===$termCount,'Retry of published draft is idempotent');
$optional=dwRequest($registrar,$route,['custom_id'=>$id,'base_term_id'=>$published['id'],'custom_required'=>'0','text'=>$run.' nové volitelné znění']+$create);
$optionalDraft=r5Location($optional,'draft_id');
dwCheck($optional['status']===303&&(int)r5Term($id)['custom_required']===1,'Draft-only required-flag change does not affect new applications');
dwCheck(dwRequest($registrar,$route,['confirmed'=>'1','draft_id'=>$optionalDraft]+$publish)['status']===303,'Publishing optional version changes future requirements');
$optionalTerm=r5Term($id);
dwCheck((int)$optionalTerm['custom_required']===0&&$optionalTerm['consent_text_plain']===$run.' nové volitelné znění','Optional consent has its own immutable text version');
$q=$pdo->prepare('SELECT * FROM club_event_term_versions WHERE id=?');$q->execute([$published['id']]);
dwCheck(r5Hash($q->fetch())===$publishedHash,'Earlier published text and metadata remain byte-for-byte unchanged');
$hide=['custom_action'=>'hide','custom_id'=>$id,'base_term_id'=>$optionalTerm['id']];
$noHide=dwRequest($registrar,$route,$hide);
dwCheck(str_contains($noHide['body'],'Potvrďte skrytí')&&(int)r5Term($id)['custom_visible']===1,'Hiding consent also requires explicit confirmation');
dwCheck(dwRequest($registrar,$route,['confirmed'=>'1']+$hide)['status']===303,'Registrar can hide consent for new applications');
dwCheck((int)r5Term($id)['custom_visible']===0&&!in_array($id,array_map('intval',array_column(clubConsentCurrent($pdo),'custom_consent_id')),true),'Hidden consent disappears from future public bundle');
$q->execute([$published['id']]);dwCheck(r5Hash($q->fetch())===$publishedHash,'Hiding never rewrites historical published text');

$person=dwInsert('sportovci',['jmeno'=>'LOCALHOST','prijmeni'=>$run,'narozeni'=>'2015-01-02','stav_clenstvi'=>'cekajici','hash'=>bin2hex(random_bytes(32)),'uci'=>0,'email'=>'']);
$program=dwInsert('club_programs',['code'=>bin2hex(random_bytes(8)),'name'=>$run.' syntetický program','created_by_trainer_id'=>$actors['registrar']]);

$missing=dwRequest($registrar,'paper_registrations_admin.php',['scope_reference'=>'']+$paper);

dwCheck(r5PaperCount()===$paperCount&&str_contains($badDate['body'],'nesmí být v budoucnu'),'Future signature date is refused without partial evidence');
dwCheck(dwRequest($finance,'paper_registrations_admin.php',$paper)['status']===403,'Finance cannot record private paper evidence with valid CSRF');

$decide=['action'=>'accepted','id'=>$record,'reason'=>$run.' výslovné ověření'];

$report=['run'=>$run,'checks'=>count($checks),'passed'=>$checks,'custom_consent_id'=>$id,'person_id'=>$person,'paper_record_ids'=>[$record,$newRecord],'production_mutations'=>false,'external_sends'=>false,'browser_execution'=>false];

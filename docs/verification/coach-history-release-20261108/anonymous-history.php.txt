<?php
declare(strict_types=1);
if (PHP_SAPI !== 'cli') { http_response_code(404); exit; }
$phase = $argv[1] ?? '';
if (!in_array($phase, ['before', 'after'], true)) { throw new RuntimeException('CLI before|after required.'); }
$out = ['at_utc' => gmdate('c'), 'phase' => $phase, 'method' => 'Anonymous coach history GET, no cookies, redirects or POST; no bodies or private headers stored.', 'cases' => []];
foreach (['ajax_treninky.php', 'moje_treninky.php'] as $path) {
    $location = null;
    $url = 'https://kis.kovopraha.cz/' . $path . '?history_page=2';
    $handle = curl_init($url);
    curl_setopt_array($handle, [CURLOPT_RETURNTRANSFER => true, CURLOPT_FOLLOWLOCATION => false,
        CURLOPT_CONNECTTIMEOUT => 15, CURLOPT_TIMEOUT => 45, CURLOPT_USERAGENT => 'KIS-authorized-read-verification',
        CURLOPT_HEADERFUNCTION => static function ($handle, string $line) use (&$location): int {
            if (stripos($line, 'Location:') === 0) { $location = parse_url(trim(substr($line, 9)), PHP_URL_PATH); }
            return strlen($line);
        }]);
    $body = curl_exec($handle);
    $status = curl_getinfo($handle, CURLINFO_RESPONSE_CODE);
    $error = curl_error($handle);
    curl_close($handle);
    if ($body === false) { throw new RuntimeException($error); }
    unset($body);
    $pass = in_array($status, [401, 403], true) || (in_array($status, [301, 302, 303, 307, 308], true)
        && is_string($location) && basename($location) === 'login.php');
    $out['cases'][] = ['url' => $url, 'status' => $status, 'redirect_path' => $location, 'pass' => $pass];
}
$out['pass'] = !in_array(false, array_column($out['cases'], 'pass'), true);
$out['limitation'] = 'Authenticated coach history production UAT NOT RUN.';
file_put_contents(__DIR__ . '/anonymous-history-' . $phase . '.json', json_encode($out, JSON_PRETTY_PRINT | JSON_UNESCAPED_SLASHES | JSON_THROW_ON_ERROR) . PHP_EOL);
echo json_encode($out, JSON_THROW_ON_ERROR) . PHP_EOL;
if (!$out['pass']) { exit(1); }

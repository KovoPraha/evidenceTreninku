<?php
require getcwd() . '/vendor/autoload.php';
function normalizeFormat($node) {
    if ($node instanceof PhpParser\Node\Stmt\InlineHTML) {
        return ['html_nonwhitespace' => preg_replace('/\s+/u', '', $node->value)];
    }
    if ($node instanceof PhpParser\Node) {
        $result = ['type' => $node->getType()];
        foreach ($node->getSubNodeNames() as $key) $result[$key] = normalizeFormat($node->$key);
        return $result;
    }
    return is_array($node) ? array_map('normalizeFormat', $node) : $node;
}
$parser = (new PhpParser\ParserFactory())->createForHostVersion();
$before = normalizeFormat($parser->parse(file_get_contents($argv[1])));
$after = normalizeFormat($parser->parse(file_get_contents('booking/eshop.php')));
if ($before !== $after) throw new RuntimeException('Formatting changed PHP AST or non-whitespace HTML.');
echo json_encode(['php_ast_identical' => true, 'inline_html_nonwhitespace_identical' => true,
    'baseline' => 'After intended helper extraction, before mandatory formatting.',
    'limit' => 'HTML whitespace is ignored here; rendered text and controls are verified separately.'], JSON_PRETTY_PRINT) . PHP_EOL;

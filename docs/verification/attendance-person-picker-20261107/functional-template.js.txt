async (page) => {
  const credential = JSON.parse('__PRIVATE_CREDENTIAL__');
  const browser = page.context().browser();
  const root = 'C:/Users/Marek/Documents/Codex/evidencePavel-attendance-picker-20261107/output/playwright/attendance-picker-20261107/';
  const base = 'http://127.0.0.1:18095';
  const report = {complete: false, contexts: []};
  for (const js of [false, true]) {
    const context = await browser.newContext({javaScriptEnabled: js, viewport: {width: 1036, height: 703}});
    const p = await context.newPage();
    const checks = [];
    const check = (ok, name) => { if (!ok) throw new Error(name); checks.push(name); };
    await p.goto(base + '/login.php');
    await p.getByRole('textbox', {name: 'Jméno nebo e-mail', exact: true}).fill(credential.login);
    await p.getByRole('textbox', {name: 'Heslo', exact: true}).fill(credential.password);
    await p.getByRole('button', {name: 'Přihlásit se', exact: true}).click();
    await p.goto(base + '/training_closure_admin.php?id=101');
    const actual = p.locator('#actual-attendance form');
    const people = actual.locator('select[name="participant_ids[]"]');
    const ids = () => people.evaluate(s => Array.from(s.selectedOptions, o => o.value));
    check((await ids()).length === 200, 'Initial 200 selected participants');
    check(await actual.locator('select[name="record_source"]').inputValue() === 'paper', 'Initial paper source');
    const version = await actual.locator('input[name="version"]').inputValue();
    const reason = 'LOCALHOST rozepsaná docházka <bez uložení>.';
    await actual.locator('textarea[name="reason"]').fill(reason);
    await actual.locator('input[name="guest_reason"]').fill('LOCALHOST rozepsaný důvod hosta.');
    await actual.locator('input[name="confirm_action"]').check();
    await actual.locator('input[name="participant_q"]').fill('20000');
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), actual.locator('input[name="participant_q"]').press('Enter')]);
    check((await ids()).length === 200, 'Keyboard Enter search preserves selections');
    check(await actual.locator('textarea[name="reason"]').inputValue() === reason, 'Search preserves draft text');
    check(await actual.locator('input[name="version"]').inputValue() === version, 'Search keeps concurrency version');
    const selected = await ids();
    check(await people.locator('option[value="20000"]').count() === 1, 'Off-page guest is searchable');
    await people.selectOption([...selected, '20000']);
    await actual.locator('input[name="participant_q"]').fill('NOT FOUND');
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), actual.getByRole('button', {name: 'Hledat sportovce', exact: true}).click()]);
    check((await ids()).length === 201 && (await ids()).includes('20000'), 'New guest pinned after empty search');
    check(await actual.locator('input[name="confirm_action"]').isChecked(), 'Confirmation remains checked');
    await people.selectOption([]);
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), actual.getByRole('button', {name: 'Hledat sportovce', exact: true}).click()]);
    check((await ids()).length === 0, 'Intentionally empty selection remains empty');
    await actual.locator('input[name="participant_q"]').fill('');
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), actual.getByRole('button', {name: 'Hledat sportovce', exact: true}).click()]);
    await actual.getByRole('button', {name: 'Další osoby', exact: true}).focus();
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), actual.getByRole('button', {name: 'Další osoby', exact: true}).press('Enter')]);
    check((await actual.getByRole('status').innerText()).includes('Strana 2'), 'Keyboard paging reaches second page');
    check((await ids()).length === 0, 'Paging keeps empty draft');
    await p.goto(base + '/training_closure_admin.php?id=101');
    const excuse = p.locator('#personal-excuse form');
    await excuse.locator('input[name="received_at"]').fill('2026-11-07T06:00');
    await excuse.locator('select[name="source"]').selectOption('paper');
    await excuse.locator('input[name="reason"]').fill('LOCALHOST osobní omluva z papíru.');
    await excuse.locator('input[name="excuse_q"]').fill('20000');
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), excuse.getByRole('button', {name: 'Hledat sportovce', exact: true}).click()]);
    await excuse.locator('select[name="person_id"]').selectOption('20000');
    await excuse.locator('input[name="excuse_q"]').fill('NOT FOUND');
    await Promise.all([p.waitForNavigation({waitUntil: 'load'}), excuse.getByRole('button', {name: 'Hledat sportovce', exact: true}).click()]);
    check(await excuse.locator('select[name="person_id"]').inputValue() === '20000', 'Excuse selection survives different query');
    check(await excuse.locator('input[name="received_at"]').inputValue() === '2026-11-07T06:00', 'Excuse time retained');
    check(await excuse.locator('select[name="source"]').inputValue() === 'paper', 'Excuse source retained');
    check(await excuse.locator('input[name="reason"]').inputValue() === 'LOCALHOST osobní omluva z papíru.', 'Excuse reason retained');
    if (js) await p.waitForFunction(() => document.body.classList.contains('app-ui-ready'));
    check(await p.locator('body').evaluate(b => b.classList.contains('app-ui-ready')) === js, 'Expected JavaScript mode');
    await p.goto(base + '/training_closure_admin.php?id=101');
    await p.screenshot({path: root + (js ? 'picker-js-desktop.png' : 'picker-nojs-desktop.png'), fullPage: true});
    await p.setViewportSize({width: 390, height: 844});
    check(await p.evaluate(() => document.documentElement.scrollWidth <= innerWidth), '390 px layout has no horizontal overflow');
    await p.screenshot({path: root + (js ? 'picker-js-mobile.png' : 'picker-nojs-mobile.png'), fullPage: true});
    report.contexts.push({javascript_enabled: js, checks, native_search_and_keyboard: true, screenshot: js ? 'picker-js-desktop.png' : 'picker-nojs-desktop.png'});
    await context.close();
  }
  report.complete = true;
  return report;
}

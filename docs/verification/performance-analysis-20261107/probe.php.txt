function connection(bool $profile): PDO {
    return $profile ? new Tests\Support\QueryProfilePdo('sqlite::memory:')
        : new PDO('sqlite::memory:', null, null, [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC]);
}
function copyFixture(PDO $source, PDO $target): void {
    foreach ($source->query("SELECT name,sql FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'")->fetchAll() as $table) {
        $target->exec($table['sql']);
        foreach ($source->query('SELECT * FROM "' . $table['name'] . '"')->fetchAll(PDO::FETCH_NUM) as $row) {
            $insert = $target->prepare('INSERT INTO "' . $table['name'] . '" VALUES(' . implode(',', array_fill(0, count($row), '?')) . ')');
            $insert->execute($row);
        }
    }
}
function observe(PDO $plain, Tests\Support\QueryProfilePdo $instrument, callable $operation): array {
    for ($i = 0; $i < 3; $i++) { $operation($plain); }
    $times = []; $peaks = [];
    for ($i = 0; $i < 20; $i++) {
        memory_reset_peak_usage(); $before = memory_get_usage(); $start = hrtime(true);
        $result = $operation($plain);
        $times[] = (hrtime(true) - $start) / 1e6;
        $peaks[] = max(0, memory_get_peak_usage() - $before);
        unset($result);
    }
    $operation($instrument); // Warm schema availability just as on the plain connection.
    $instrument->profile = []; $instrument->profiling = true;
    $actual = $operation($instrument);
    $instrument->profiling = false;
    $expected = $operation($plain);
    if ($actual !== $expected) { throw new RuntimeException('Instrumented data mismatch'); }
    $queries = [];
    foreach ($instrument->profile as $entry) {
        $sql = preg_replace('/\s+/', ' ', trim($entry['sql']));
        $queries[$sql] = ($queries[$sql] ?? 0) + 1;
    }
    $ordered = $times; sort($ordered);
    return ['sql' => count($instrument->profile), 'fetched_rows' => array_sum(array_column($instrument->profile, 'fetched_rows')),
        'median_ms' => ($ordered[9] + $ordered[10]) / 2, 'p95_ms' => $ordered[18], 'peak_increment_bytes' => max($peaks),
        'output_items' => is_array($actual) ? count($actual) : null, 'samples_ms' => $times, 'queries' => $queries, 'parity' => true];
}
